const { expect } = require('@playwright/test');
const BASE = 'http://127.0.0.1:8766';
function assertQaURL(baseURL) {
  expect(baseURL, 'FAIL CLOSED: Tests must target only the restored local QA service').toBe(BASE);
}
async function fence(page) {
  const offsite = [];
  await page.route('**/*', route => {
    const url = new URL(route.request().url());
    if (url.origin !== BASE) {
      offsite.push(url.origin); // no user data, query strings or credentials
      return route.abort();
    }
    return route.continue();
  });
  return offsite;
}
async function assertWorkspaceHealthy(page) {
  await expect(page.locator('#root')).toBeVisible();
  // A painted error boundary is NOT a successful login screen.
  await expect(page.getByText('Unable to restore your workspace')).toHaveCount(0);
  await expect(page.getByText(/Cannot read properties of undefined/i)).toHaveCount(0);
  await expect(page.getByText(/SESSION CHECK/i)).toHaveCount(0);
  const text = (await page.locator('#root').innerText()).trim();
  expect(text.length, 'Root has no meaningful text').toBeGreaterThan(10);
}
function assertSessionEnvelope(payload) {
  expect(payload && typeof payload === 'object', 'Session must be an object').toBeTruthy();
  expect(payload.data && typeof payload.data === 'object', 'Expected JSON data envelope').toBeTruthy();
  expect(Array.isArray(payload.data.organizations), 'Session must contain organizations array').toBe(true);
  expect(typeof payload.data.csrf, 'Session must contain CSRF token').toBe('string');
  expect(payload.data.csrf.length, 'CSRF token must be nonempty').toBeGreaterThan(12);
}
module.exports = { BASE, assertQaURL, fence, assertWorkspaceHealthy, assertSessionEnvelope };
